#!/bin/sh
# Copie para .githooks/pre-push, junto de config.sh.
[ "${SKIP_GIT_HOOKS:-0}" = 1 ] && exit 0
. "$(dirname "$0")/config.sh" || exit 1
if [ "${ALLOW_DIRECT_PUSH:-0}" != 1 ]; then
  case "$PROTECTED_BRANCHES" in
    ''|*'{{'*) hook_error 'configure PROTECTED_BRANCHES em .githooks/config.sh.' ;;
  esac
fi
UPDATES=0
while IFS=' ' read -r local_ref local_sha remote_ref remote_sha; do
  [ -n "$remote_ref" ] || continue
  if [ "${ALLOW_DIRECT_PUSH:-0}" != 1 ]; then
    for branch in $PROTECTED_BRANCHES; do
      if [ "$remote_ref" = "refs/heads/$branch" ]; then
        hook_error "push para $branch recusado; envie a branch de trabalho e abra um PR. Em emergência: ALLOW_DIRECT_PUSH=1 (as verificações continuam)."
      fi
    done
  fi
  # Ref excluída: o object ID local é composto apenas de zeros, inclusive com SHA-256.
  case "$local_sha" in *[!0]*) UPDATES=1 ;; esac
done
[ "$UPDATES" = 1 ] || exit 0
# stdin do hook contém refs, não deve ser consumido por um comando de teste.
hook_run_script hooks:pre-push < /dev/null
