#!/bin/sh
# Copie para .githooks/prepare-commit-msg, junto de config.sh.
[ "${SKIP_GIT_HOOKS:-0}" = 1 ] && exit 0
MSG_FILE=${1:?arquivo de mensagem obrigatório}
SOURCE=${2:-}
case "$SOURCE" in merge|squash|commit) exit 0 ;; esac
. "$(dirname "$0")/config.sh" || exit 1
[ -f "$MSG_FILE" ] || hook_error 'arquivo de mensagem inexistente.'
MESSAGE=$(git stripspace --strip-comments < "$MSG_FILE") || exit 1
SUBJECT=${MESSAGE%%'
'*}
case "$SUBJECT" in 'Revert "'*|'fixup! '*|'squash! '*|'amend! '*) exit 0 ;; esac
BRANCH=$(git symbolic-ref --quiet --short HEAD 2>/dev/null) || exit 0
hook_check_prefix
hook_has_key "$MESSAGE" && exit 0

PREFIX=${BRANCH%%/*}
case "$PREFIX" in
  feat|feature) TYPE=feat ;;
  fix|bugfix) TYPE=fix ;;
  hotfix|chore|docs|refactor|test|style|perf|build|ci|revert) TYPE=$PREFIX ;;
  *) exit 0 ;;
esac
[ "$PREFIX" != "$BRANCH" ] || exit 0
REST=${BRANCH#*/}
KEY=$(printf '%s\n' "$REST" | awk -v prefix="$ISSUE_PREFIX" '
  {
    value = toupper($0)
    if (value !~ ("^" prefix "-[1-9][0-9]*(-[A-Z0-9]+)*$")) next
    match(value, "^" prefix "-[1-9][0-9]*")
    key = substr(value, RSTART, RLENGTH)
    rest = substr(value, RLENGTH + 1)
    if (rest ~ ("(^|[^A-Z0-9_])" prefix "-[1-9][0-9]*([^A-Z0-9_]|$)")) next
    print key
  }
') || exit 1
[ -n "$KEY" ] || exit 0

# Temporário no diretório da mensagem: mesma partição para o rename final.
case "$MSG_FILE" in /*|./*|../*|?:/*) ;; *) MSG_FILE=./$MSG_FILE ;; esac
TMP=$(mktemp "${MSG_FILE}.XXXXXX") || exit 1
trap 'rm -f "$TMP"' 0
trap 'exit 1' 1 2 3 15
TYPES='feat|fix|hotfix|chore|docs|refactor|test|style|perf|build|ci|revert'
CONVENTIONAL="^($TYPES)(\([^()]+\))?!?: "
if [ -z "$MESSAGE" ]; then
  { printf '%s(%s): \n' "$TYPE" "$KEY"; cat "$MSG_FILE"; } > "$TMP" || exit 1
elif printf '%s\n' "$SUBJECT" | grep -qE "$CONVENTIONAL" &&
     ! printf '%s\n' "$SUBJECT" | grep -qE '^[a-z]+\('; then
  COMMENT=$(git config --get core.commentChar || printf '#')
  # core.commentChar=auto deixa o Git escolher; prefira uma configuração explícita.
  [ "$COMMENT" != auto ] || hook_error 'defina core.commentChar explicitamente para usar a injeção.'
  awk -v key="$KEY" -v comment="$COMMENT" '
    !done && $0 !~ /^[[:space:]]*$/ && substr($0, 1, length(comment)) != comment {
      colon = index($0, ": ")
      header = substr($0, 1, colon - 1)
      bang = ""
      if (substr(header, length(header), 1) == "!") {
        header = substr(header, 1, length(header) - 1)
        bang = "!"
      }
      $0 = header "(" key ")" bang substr($0, colon)
      done = 1
    }
    { print }
  ' "$MSG_FILE" > "$TMP" || exit 1
else
  # Preserva o escopo existente e insere o trailer antes dos comentários/diffs.
  git interpret-trailers --trailer "Refs: $KEY" < "$MSG_FILE" > "$TMP" || exit 1
fi
mv -f "$TMP" "$MSG_FILE" || exit 1
exit 0
