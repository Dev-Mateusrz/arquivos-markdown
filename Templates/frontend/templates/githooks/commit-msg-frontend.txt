#!/bin/sh
# Copie para .githooks/commit-msg, junto de config.sh. Não altera a mensagem.
[ "${SKIP_GIT_HOOKS:-0}" = 1 ] && exit 0
. "$(dirname "$0")/config.sh" || exit 1
MSG_FILE=${1:?arquivo de mensagem obrigatório}
[ -f "$MSG_FILE" ] || hook_error 'arquivo de mensagem inexistente.'
MESSAGE=$(git stripspace --strip-comments < "$MSG_FILE") || exit 1
SUBJECT=${MESSAGE%%'
'*}

# O hook não recebe o SOURCE de prepare-commit-msg. Reconhece o estado do Git.
GIT_DIR=$(git rev-parse --absolute-git-dir) || exit 1
if [ -f "$GIT_DIR/MERGE_HEAD" ] || [ -f "$GIT_DIR/REVERT_HEAD" ] || [ -f "$GIT_DIR/SQUASH_MSG" ]; then
  exit 0
fi
case "$SUBJECT" in
  'Revert "'*|'fixup! '*|'squash! '*|'amend! '*) exit 0 ;;
esac

hook_check_prefix
[ -n "$SUBJECT" ] || hook_error 'mensagem de commit vazia.'
# Caracteres conforme a locale; não equivale a contar grafemas Unicode.
[ "${#SUBJECT}" -le 100 ] || hook_error 'a primeira linha passou do limite de 100 caracteres da locale.'
TYPES='feat|fix|hotfix|chore|docs|refactor|test|style|perf|build|ci|revert'
if ! printf '%s\n' "$SUBJECT" | grep -qE "^($TYPES)(\([^()]+\))?!?: [^[:space:]].{3,}$"; then
  hook_error 'use <tipo>(escopo opcional): descrição de pelo menos 4 caracteres; ! é permitido antes de :.'
fi
hook_has_key "$MESSAGE" || hook_error "falta uma chave válida, como ${ISSUE_PREFIX}-42, no título ou corpo."
exit 0
